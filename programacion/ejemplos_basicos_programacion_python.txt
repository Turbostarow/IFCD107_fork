26. PROGRAMA DE CLASIFICACIÓN DE COMPRAS
----------------

compras = [120, 350, 80, 420]

    if compras[i] >= 300:
        categoria = "Alta"
    elif compras[i] >= 100:
        categoria = "Media"
    else:
        categoria = "Baja"

    print(clientes[i], compras[i], categoria)

28. PEQUEÑO PROYECTO: CALCULADORA
----------------
def calculadora():

    a = float(input("Primer número: "))
    operador = input("Operador (+, -, *, /): ")
    b = float(input("Segundo número: "))

    if operador == "+":
        resultado = a + b

    elif operador == "-":
        resultado = a - b

    elif operador == "*":
        resultado = a * b

    elif operador == "/":
        if b != 0:
            resultado = a / b
        else:
            print("No se puede dividir entre cero")
            return

    else:
        print("Operador no válido")
        return

    print("Resultado:", resultado)


calculadora()
